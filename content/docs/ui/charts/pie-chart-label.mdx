```tsx title="components/charts/pie-chart-label.tsx"
"use client"

import { Pie, PieChart } from "recharts"

import {
  Card,
  CardContent,
  CardDescription,
  CardHeader,
  CardTitle,
} from "@/components/ui/card"
import {
  type ChartConfig,
  ChartContainer,
  ChartTooltip,
  ChartTooltipContent,
} from "@/components/ui/chart"

const chartData = [
  { device: "phone", users: 275, fill: "var(--color-phone)" },
  { device: "laptop", users: 200, fill: "var(--color-laptop)" },
  { device: "tablet", users: 187, fill: "var(--color-tablet)" },
  { device: "watch", users: 173, fill: "var(--color-watch)" },
  { device: "other", users: 90, fill: "var(--color-other)" },
]

const chartConfig = {
  users: { label: "Users" },
  phone: { label: "Phone", color: "var(--blue)" },
  laptop: { label: "Laptop", color: "var(--teal)" },
  tablet: { label: "Tablet", color: "var(--indigo)" },
  watch: { label: "Watch", color: "var(--orange)" },
  other: { label: "Other", color: "var(--gray)" },
} satisfies ChartConfig

export function PieChartLabel() {
  return (
    <Card variant="filled" size="sm">
      <CardHeader>
        <CardTitle>Labeled</CardTitle>
        <CardDescription>Each name sits outside its slice</CardDescription>
      </CardHeader>
      <CardContent>
        <ChartContainer
          config={chartConfig}
          className="aspect-auto h-52 w-full"
        >
          <PieChart accessibilityLayer>
            <ChartTooltip
              cursor={false}
              content={<ChartTooltipContent hideLabel nameKey="device" />}
            />
            <Pie
              data={chartData}
              dataKey="users"
              nameKey="device"
              outerRadius="68%"
              stroke="var(--surface-secondary)"
              strokeWidth={2}
              labelLine={{ stroke: "var(--separator-strong)" }}
              label={({ x, y, cx, name }) => (
                <text
                  x={x}
                  y={y}
                  textAnchor={x > cx ? "start" : "end"}
                  dominantBaseline="central"
                  className="fill-label text-xs font-semibold"
                >
                  {chartConfig[name as keyof typeof chartConfig].label}
                </text>
              )}
            />
          </PieChart>
        </ChartContainer>
      </CardContent>
    </Card>
  )
}
```

</Example>

Uses `PieChart` with one `Pie` at `outerRadius="68%"` to leave room for labels, a 2-pixel `var(--surface-secondary)` stroke, `labelLine` stroked with `var(--separator-strong)`, and a `label` render function that returns a `text` element anchored to `start` or `end` depending on which side of the center it falls. The label text is the slice's `label` from `chartConfig`. `chartData` holds five `{ device, users, fill }` records, where each `fill` is `var(--color-<device>)`. `chartConfig` defines `users` (label only) plus `phone`, `laptop`, `tablet`, `watch` and `other` colored `var(--blue)`, `var(--teal)`, `var(--indigo)`, `var(--orange)` and `var(--gray)`. The tooltip uses `ChartTooltipContent` with `hideLabel` and `nameKey="device"`.

## Usage

```tsx
import { PieChartLabel } from "@/components/charts/pie-chart-label"
```

```tsx
<PieChartLabel />
```

Replace `chartData` with your categories, giving each record a `fill` of `var(--color-<key>)`, and add a matching `chartConfig` entry with `label` and `color` for every key. Keep `dataKey` and `nameKey` on `Pie` pointed at your value and category fields, and update the tooltip's `nameKey` if you rename the category field. Labels read from `chartConfig`, so set the `label` you want displayed there. Return a value or percentage from the `label` function instead if you prefer numbers, and lower `outerRadius` if long labels are clipped.

## Notes

Label text uses the `fill-label` class so it follows the label color in light and dark appearance. `accessibilityLayer` is set on `PieChart`, so the chart is focusable and arrow keys move between slices.
