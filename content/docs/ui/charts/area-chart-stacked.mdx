```tsx title="components/charts/area-chart-stacked.tsx"
import { Area, AreaChart, CartesianGrid, XAxis } from "recharts"

import {
  Card,
  CardContent,
  CardDescription,
  CardHeader,
  CardTitle,
} from "@/components/ui/card"

  type ChartConfig,
  ChartContainer,
  ChartLegend,
  ChartLegendContent,
  ChartTooltip,
  ChartTooltipContent,
} from "@/components/ui/chart"

const chartData = [
  { month: "January", desktop: 186, mobile: 80 },
  { month: "February", desktop: 305, mobile: 200 },
  { month: "March", desktop: 237, mobile: 120 },
  { month: "April", desktop: 173, mobile: 190 },
  { month: "May", desktop: 209, mobile: 130 },
  { month: "June", desktop: 264, mobile: 140 },
]

const chartConfig = {
  desktop: { label: "Desktop", color: "var(--blue)" },
  mobile: { label: "Mobile", color: "var(--teal)" },
} satisfies ChartConfig

export function AreaChartStacked() {
  return (
    <Card variant="filled" size="sm">
      <CardHeader>
        <CardTitle>Stacked</CardTitle>
        <CardDescription>Desktop and mobile combined</CardDescription>
      </CardHeader>
      <CardContent>
        <ChartContainer
          config={chartConfig}
          className="aspect-auto h-48 w-full"
        >
          <AreaChart
            accessibilityLayer
            data={chartData}
            margin={{ top: 10, right: 12, left: 12, bottom: 0 }}
          >
            <CartesianGrid vertical={false} strokeDasharray="2 4" />
            <XAxis
              dataKey="month"
              tickLine={false}
              axisLine={false}
              tickMargin={8}
              interval={0}
              tickFormatter={(value: string) => value.slice(0, 3)}
            />
            <ChartTooltip content={<ChartTooltipContent indicator="dot" />} />
            <ChartLegend content={<ChartLegendContent />} />
            <Area
              dataKey="mobile"
              type="natural"
              stackId="a"
              stroke="var(--color-mobile)"
              strokeWidth={2}
              fill="var(--color-mobile)"
              fillOpacity={0.2}
            />
            <Area
              dataKey="desktop"
              type="natural"
              stackId="a"
              stroke="var(--color-desktop)"
              strokeWidth={2}
              fill="var(--color-desktop)"
              fillOpacity={0.2}
            />
          </AreaChart>
        </ChartContainer>
      </CardContent>
    </Card>
  )
}
```

</Example>

Uses `AreaChart` with two `Area` series sharing `stackId="a"`: `mobile` forms the base layer and `desktop` stacks on top, both `type="natural"` at `fillOpacity={0.2}`. `chartData` holds six months of `{ month, desktop, mobile }` records; `chartConfig` colors `desktop` with `var(--blue)` and `mobile` with `var(--teal)`. `ChartLegend` with `ChartLegendContent` labels both series, and the tooltip uses `indicator="dot"`.

## Usage

```tsx
import { AreaChartStacked } from "@/components/charts/area-chart-stacked"
```

```tsx
<AreaChartStacked />
```

Replace `chartData` keeping one category key and one numeric key per series. To add a layer, add a `chartConfig` entry with `label` and `color`, then an `Area` with the same `stackId`. Series are stacked in the order the `Area` elements appear, so list the base layer first.

## Notes

The legend reads labels and colors from `chartConfig`. `accessibilityLayer` is set on the chart, so the plot is focusable and arrow keys move the tooltip between months.
