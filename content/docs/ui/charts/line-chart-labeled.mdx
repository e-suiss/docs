```tsx title="components/charts/line-chart-labeled.tsx"
import { LabelList, Line, LineChart, XAxis } from "recharts"

  type ChartConfig,
  ChartContainer,
  ChartTooltip,
  ChartTooltipContent,
} from "@/components/ui/chart"

const chartData = [
  { month: "January", desktop: 186, mobile: 80 },
  { month: "February", desktop: 305, mobile: 200 },
  { month: "March", desktop: 237, mobile: 120 },
  { month: "April", desktop: 173, mobile: 190 },
  { month: "May", desktop: 209, mobile: 130 },
  { month: "June", desktop: 264, mobile: 140 },
]

const chartConfig = {
  desktop: { label: "Desktop", color: "var(--blue)" },
} satisfies ChartConfig

export function LineChartLabeled() {
  return (
    <Card variant="filled" size="sm">
      <CardHeader>
        <CardTitle>Labeled</CardTitle>
        <CardDescription>Every point shows its value</CardDescription>
      </CardHeader>
      <CardContent>
        <ChartContainer
          config={chartConfig}
          className="aspect-auto h-48 w-full"
        >
          <LineChart
            accessibilityLayer
            data={chartData}
            margin={{ top: 24, right: 16, left: 16, bottom: 0 }}
          >
            <XAxis
              dataKey="month"
              tickLine={false}
              axisLine={false}
              tickMargin={8}
              interval={0}
              tickFormatter={(value: string) => value.slice(0, 3)}
            />
            <ChartTooltip content={<ChartTooltipContent hideLabel />} />
            <Line
              dataKey="desktop"
              type="natural"
              stroke="var(--color-desktop)"
              strokeWidth={2.5}
              dot={{ r: 4, fill: "var(--color-desktop)", strokeWidth: 0 }}
            >
              <LabelList
                position="top"
                offset={10}
                className="fill-label font-semibold"
                fontSize={12}
              />
            </Line>
          </LineChart>
        </ChartContainer>
      </CardContent>
    </Card>
  )
}
```

</Example>

Uses `LineChart` with one `Line` (`type="natural"`, `strokeWidth={2.5}`) whose `dot` draws a 4-pixel filled marker, and a nested `LabelList` with `position="top"`, `offset={10}` and `fontSize={12}`. There is no `CartesianGrid`, and the top margin is 24 to leave room for labels. `chartData` holds six months of `{ month, desktop, mobile }` records; only `desktop` is plotted with `chartConfig.desktop` (`var(--blue)`). The tooltip uses `hideLabel`.

## Usage

```tsx
import { LineChartLabeled } from "@/components/charts/line-chart-labeled"
```

```tsx
<LineChartLabeled />
```

Replace `chartData` and rename the `chartConfig` entry to match your series. Pass a `formatter` to `LabelList` to add units, or change `position` to place labels below points. Increase the `margin` if labels are clipped at the edges.

## Notes

Labels use the `fill-label` class so they follow the label color in light and dark appearance. `accessibilityLayer` is set on `LineChart`, so the plot is focusable and arrow keys move the tooltip between points.
