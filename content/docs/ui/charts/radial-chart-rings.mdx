```tsx title="components/charts/radial-chart-rings.tsx"
"use client"

import {
  Card,
  CardContent,
  CardDescription,
  CardHeader,
  CardTitle,
} from "@/components/ui/card"

const rings = [
  { key: "move", label: "Move", value: 720, goal: 600, color: "var(--red)" },
  {
    key: "exercise",
    label: "Exercise",
    value: 62,
    goal: 30,
    color: "var(--green)",
  },
  { key: "stand", label: "Stand", value: 12, goal: 12, color: "var(--cyan)" },
]

const SIZE = 200
const BAR = 16

export function RadialChartRings() {
  return (
    <Card variant="filled" size="sm">
      <CardHeader>
        <CardTitle>Activity rings</CardTitle>
        <CardDescription>Move, exercise and stand</CardDescription>
      </CardHeader>
      <CardContent className="flex justify-center">
        <svg
          role="img"
          aria-label={rings
            .map((ring) => `${ring.label} ${ring.value} of ${ring.goal}`)
            .join(", ")}
          viewBox={`0 0 ${SIZE} ${SIZE}`}
          className="size-50 -rotate-90"
        >
          {rings.map((ring, index) => {
            const radius = SIZE / 2 - BAR / 2 - index * (BAR + 2)
            const circumference = 2 * Math.PI * radius
            const progress = Math.min(1, ring.value / ring.goal)
            return (
              <g key={ring.key}>
                <circle
                  cx={SIZE / 2}
                  cy={SIZE / 2}
                  r={radius}
                  fill="none"
                  strokeWidth={BAR}
                  stroke={`color-mix(in oklab, ${ring.color} 22%, transparent)`}
                />
                <circle
                  cx={SIZE / 2}
                  cy={SIZE / 2}
                  r={radius}
                  fill="none"
                  strokeWidth={BAR}
                  strokeLinecap="round"
                  stroke={ring.color}
                  strokeDasharray={`${progress * circumference} ${circumference}`}
                />
              </g>
            )
          })}
        </svg>
      </CardContent>
    </Card>
  )
}
```

</Example>

This variant doesn't use Recharts or `ChartContainer`. It renders an `svg` rotated by -90 degrees with one pair of `circle` elements per entry in `rings`: a track stroked with a 22% `color-mix` tint and a progress arc whose length is set with `strokeDasharray`. Each ring entry is `{ key, label, value, goal, color }`, using `var(--red)`, `var(--green)` and `var(--cyan)`. Progress is `value / goal`, capped at one full circle. `SIZE` and `BAR` set the viewBox and stroke width.

## Usage

```tsx
import { RadialChartRings } from "@/components/charts/radial-chart-rings"
```

```tsx
<RadialChartRings />
```

Replace the `rings` values and goals with your own, or add entries with a `color` token; each additional ring is placed further inward. Change `SIZE` and `BAR` to resize the rings and their thickness.

## Notes

The `svg` has `role="img"` and an `aria-label` built from each ring's label, value and goal. Values above the goal render as a full ring; there is no overlap lap.
