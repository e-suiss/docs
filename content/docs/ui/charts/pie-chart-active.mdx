```tsx title="components/charts/pie-chart-active.tsx"
"use client"

import { Pie, PieChart, Sector } from "recharts"

import {
  Card,
  CardContent,
  CardDescription,
  CardHeader,
  CardTitle,
} from "@/components/ui/card"
import {
  type ChartConfig,
  ChartContainer,
  ChartTooltip,
  ChartTooltipContent,
} from "@/components/ui/chart"

const chartData = [
  { device: "phone", users: 275, fill: "var(--color-phone)" },
  { device: "laptop", users: 200, fill: "var(--color-laptop)" },
  { device: "tablet", users: 187, fill: "var(--color-tablet)" },
  { device: "watch", users: 173, fill: "var(--color-watch)" },
  { device: "other", users: 90, fill: "var(--color-other)" },
]

const chartConfig = {
  users: { label: "Users" },
  phone: { label: "Phone", color: "var(--blue)" },
  laptop: { label: "Laptop", color: "var(--teal)" },
  tablet: { label: "Tablet", color: "var(--indigo)" },
  watch: { label: "Watch", color: "var(--orange)" },
  other: { label: "Other", color: "var(--gray)" },
} satisfies ChartConfig

const ACTIVE = 0

export function PieChartActive() {
  return (
    <Card variant="filled" size="sm">
      <CardHeader>
        <CardTitle>Highlighted slice</CardTitle>
        <CardDescription>The selected slice pops out</CardDescription>
      </CardHeader>
      <CardContent>
        <ChartContainer
          config={chartConfig}
          className="aspect-auto h-52 w-full"
        >
          <PieChart accessibilityLayer>
            <ChartTooltip
              cursor={false}
              content={<ChartTooltipContent hideLabel nameKey="device" />}
            />
            <Pie
              data={chartData}
              dataKey="users"
              nameKey="device"
              innerRadius="50%"
              outerRadius="78%"
              strokeWidth={0}
              shape={(props, index) =>
                index === ACTIVE ? (
                  <g>
                    <Sector
                      {...props}
                      outerRadius={(props.outerRadius ?? 0) + 10}
                    />
                    <Sector
                      {...props}
                      innerRadius={(props.outerRadius ?? 0) + 13}
                      outerRadius={(props.outerRadius ?? 0) + 16}
                    />
                  </g>
                ) : (
                  <Sector {...props} />
                )
              }
            />
          </PieChart>
        </ChartContainer>
      </CardContent>
    </Card>
  )
}
```

</Example>

Uses `PieChart` with one `Pie` (`innerRadius="50%"`, `outerRadius="78%"`, `strokeWidth={0}`) and a `shape` render function. For the slice whose index equals `ACTIVE`, it returns two `Sector` elements: the slice extended by 10 pixels and a thin arc 13 to 16 pixels beyond the original radius; other slices render a plain `Sector`. `chartData` holds five `{ device, users, fill }` records, where each `fill` is `var(--color-<device>)`. `chartConfig` defines `users` (label only) plus `phone`, `laptop`, `tablet`, `watch` and `other` colored `var(--blue)`, `var(--teal)`, `var(--indigo)`, `var(--orange)` and `var(--gray)`. The tooltip uses `ChartTooltipContent` with `hideLabel` and `nameKey="device"`.

## Usage

```tsx
import { PieChartActive } from "@/components/charts/pie-chart-active"
```

```tsx
<PieChartActive />
```

Replace `chartData` with your categories, giving each record a `fill` of `var(--color-<key>)`, and add a matching `chartConfig` entry with `label` and `color` for every key. Keep `dataKey` and `nameKey` on `Pie` pointed at your value and category fields, and update the tooltip's `nameKey` if you rename the category field. Set `ACTIVE` to the index of the slice to highlight, or replace it with a value from `React.useState` to drive the highlight from user input. Change the pixel offsets inside `shape` to adjust how far the slice and ring extend.

## Notes

The highlight is static: `ACTIVE` is a constant, so it doesn't follow hover. `accessibilityLayer` is set on `PieChart`, so the chart is focusable and arrow keys move between slices.
