```tsx title="components/charts/tooltip-chart-label-formatter.tsx"
"use client"

import * as React from "react"
import { Area, AreaChart, XAxis } from "recharts"

import {
  Card,
  CardContent,
  CardDescription,
  CardHeader,
  CardTitle,
} from "@/components/ui/card"
import {
  type ChartConfig,
  ChartContainer,
  ChartTooltip,
  ChartTooltipContent,
} from "@/components/ui/chart"

function dayKey(date: Date) {
  return [
    date.getFullYear(),
    String(date.getMonth() + 1).padStart(2, "0"),
    String(date.getDate()).padStart(2, "0"),
  ].join("-")
}

const chartData = [
  512, 604, 455, 389, 662, 598, 431, 520, 690, 447, 575, 402, 633, 558,
].map((energy, index) => ({
  date: dayKey(new Date(2026, 8, 23 + index)),
  energy,
}))

const chartConfig = {
  energy: { label: "Active energy", color: "var(--red)" },
} satisfies ChartConfig

const longDate = new Intl.DateTimeFormat("en-US", {
  weekday: "long",
  month: "long",
  day: "numeric",
})

export function TooltipChartLabelFormatter() {
  const id = React.useId()
  return (
    <Card variant="filled" size="sm">
      <CardHeader>
        <CardTitle>Custom label</CardTitle>
        <CardDescription>The date written out in full</CardDescription>
      </CardHeader>
      <CardContent>
        <ChartContainer
          config={chartConfig}
          className="aspect-auto h-52 w-full"
        >
          <AreaChart
            accessibilityLayer
            data={chartData}
            margin={{ top: 70, right: 12, left: 12, bottom: 0 }}
          >
            <defs>
              <linearGradient id={id} x1="0" y1="0" x2="0" y2="1">
                <stop
                  offset="0%"
                  stopColor="var(--color-energy)"
                  stopOpacity={0.38}
                />
                <stop
                  offset="100%"
                  stopColor="var(--color-energy)"
                  stopOpacity={0.02}
                />
              </linearGradient>
            </defs>
            <XAxis
              dataKey="date"
              tickLine={false}
              axisLine={false}
              tickMargin={8}
              interval={2}
              tickFormatter={(value: string) => String(Number(value.slice(8)))}
            />
            <ChartTooltip
              defaultIndex={8}
              content={
                <ChartTooltipContent
                  indicator="line"
                  labelFormatter={(_, payload) =>
                    longDate.format(
                      new Date(`${payload[0]?.payload.date}T12:00:00`)
                    )
                  }
                />
              }
            />
            <Area
              dataKey="energy"
              type="monotone"
              stroke="var(--color-energy)"
              strokeWidth={2}
              fill={`url(#${id})`}
            />
          </AreaChart>
        </ChartContainer>
      </CardContent>
    </Card>
  )
}
```

</Example>

Uses `AreaChart` with one `Area` (`type="monotone"`) filled by a vertical `linearGradient` whose `id` comes from `React.useId()`. `chartData` holds fourteen `{ date, energy }` records with `YYYY-MM-DD` dates; the `XAxis` shows the day number for every third point (`interval={2}`). `chartConfig.energy` uses `var(--red)`. The tooltip is `ChartTooltipContent` with `indicator="line"` and a `labelFormatter` that formats the hovered record's `date` with `Intl.DateTimeFormat` (`weekday`, `month` and `day` in long form). `ChartTooltip` sets `defaultIndex={8}`.

## Usage

```tsx
import { TooltipChartLabelFormatter } from "@/components/charts/tooltip-chart-label-formatter"
```

```tsx
<TooltipChartLabelFormatter />
```

Replace `chartData` with your dated records, keeping ISO date strings so both the axis and `labelFormatter` can parse them. Change the options passed to `Intl.DateTimeFormat` in `longDate` to adjust the heading format. Remove `defaultIndex` if the tooltip should appear only on interaction.

## Notes

`labelFormatter` reads the date from `payload[0].payload` rather than its first argument, so it works regardless of what the axis displays. `accessibilityLayer` is set on the chart, so the plot is focusable and arrow keys move the tooltip between categories.
