```tsx title="components/charts/line-chart-goal.tsx"
  ChartContainer,
  ChartTooltip,
  ChartTooltipContent,

export function LineChartGoal() {
  return (
    <Card variant="filled" size="sm">
      <CardHeader>

        <CardDescription>A daily goal of 10,000 steps</CardDescription>
      </CardHeader>
      <CardContent>
        <ChartContainer
          config={chartConfig}
          className="aspect-auto h-48 w-full"

          <LineChart
            accessibilityLayer
            data={chartData}
            margin={{ top: 10, right: 4, left: 12, bottom: 0 }}

            <CartesianGrid vertical={false} strokeDasharray="2 4" />

              dataKey="day"
              tickLine={false}
              axisLine={false}
              tickMargin={8}
              interval={0}
            />

              orientation="right"
              tickLine={false}
              axisLine={false}
              tickMargin={6}
              width="auto"
              tickFormatter={(value: number) => `${value / 1000}k`}
            />
            <ReferenceLine
              y={GOAL}
              stroke="var(--green)"
              strokeDasharray="4 4"
              label={{
                value: "Goal",
                position: "insideTopLeft",
                className: "fill-label-secondary text-xs font-semibold",

            />
            <ChartTooltip
              content={

                  hideIndicator
                  formatter={(value) => (
                    <span className="font-semibold tabular-nums">
                      {stepsFormat.format(Number(value))} steps

                />
              }
            />
            <Line
              dataKey="steps"
              type="monotone"
              stroke="var(--color-steps)"
              strokeWidth={2.5}
              dot={{ r: 3.5, fill: "var(--color-steps)", strokeWidth: 0 }}
            />
          </LineChart>
        </ChartContainer>
      </CardContent>
    </Card>
  )
}
```

Uses `LineChart` with one `Line` (`type="monotone"`, `strokeWidth={2.5}`, 3.5-pixel dots) and a `ReferenceLine` at `y={GOAL}` drawn in `var(--green)` with `strokeDasharray="4 4"` and a `label` reading Goal at `insideTopLeft`. `chartData` holds seven `{ day, steps }` records, and `chartConfig.steps` uses `var(--orange)`. ## Usage

```tsx
import { LineChartGoal } from "@/components/charts/line-chart-goal"
```

```tsx
<LineChartGoal />
```

Replace `chartData` with your daily values and set `GOAL` to your target. The tooltip uses `hideIndicator` and a `formatter` that prints the value with `Intl.NumberFormat` followed by `steps`; adjust it for your unit, and change the `YAxis` `tickFormatter` if your values aren't in the thousands.

## Notes

`accessibilityLayer` is set on `LineChart`, so the plot is focusable and arrow keys move the tooltip between days.
