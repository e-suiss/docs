```tsx title="components/charts/pie-chart-donut-text.tsx"
"use client"

import { Pie, PieChart } from "recharts"

import {
  Card,
  CardContent,
  CardDescription,
  CardHeader,
  CardTitle,
} from "@/components/ui/card"
import {
  type ChartConfig,
  ChartContainer,
  ChartTooltip,
  ChartTooltipContent,
} from "@/components/ui/chart"

const chartData = [
  { device: "phone", users: 275, fill: "var(--color-phone)" },
  { device: "laptop", users: 200, fill: "var(--color-laptop)" },
  { device: "tablet", users: 187, fill: "var(--color-tablet)" },
  { device: "watch", users: 173, fill: "var(--color-watch)" },
  { device: "other", users: 90, fill: "var(--color-other)" },
]

const chartConfig = {
  users: { label: "Users" },
  phone: { label: "Phone", color: "var(--blue)" },
  laptop: { label: "Laptop", color: "var(--teal)" },
  tablet: { label: "Tablet", color: "var(--indigo)" },
  watch: { label: "Watch", color: "var(--orange)" },
  other: { label: "Other", color: "var(--gray)" },
} satisfies ChartConfig

const total = chartData.reduce((sum, item) => sum + item.users, 0)

export function PieChartDonutText() {
  return (
    <Card variant="filled" size="sm">
      <CardHeader>
        <CardTitle>Total in the middle</CardTitle>
        <CardDescription>A summary inside the ring</CardDescription>
      </CardHeader>
      <CardContent>
        <div className="relative">
          <ChartContainer
            config={chartConfig}
            className="aspect-auto h-52 w-full"
          >
            <PieChart accessibilityLayer>
              <ChartTooltip
                cursor={false}
                content={<ChartTooltipContent hideLabel nameKey="device" />}
              />
              <Pie
                data={chartData}
                dataKey="users"
                nameKey="device"
                innerRadius="62%"
                outerRadius="88%"
                strokeWidth={0}
                paddingAngle={1}
                cornerRadius={3}
              />
            </PieChart>
          </ChartContainer>
          <div className="pointer-events-none absolute inset-0 flex flex-col items-center justify-center text-center">
            <span className="text-2xl font-semibold tracking-tight tabular-nums">
              {total.toLocaleString("en-US")}
            </span>
            <span className="text-xs text-label-secondary">users</span>
          </div>
        </div>
      </CardContent>
    </Card>
  )
}
```

</Example>

Uses `PieChart` with one `Pie` (`innerRadius="62%"`, `outerRadius="88%"`, `paddingAngle={1}`, `cornerRadius={3}`, `strokeWidth={0}`). `chartData` holds five `{ device, users, fill }` records, where each `fill` is `var(--color-<device>)`. `chartConfig` defines `users` (label only) plus `phone`, `laptop`, `tablet`, `watch` and `other` colored `var(--blue)`, `var(--teal)`, `var(--indigo)`, `var(--orange)` and `var(--gray)`. The center text is a separate, absolutely positioned element over `ChartContainer` showing `total`, which is summed from `chartData` and formatted with `toLocaleString`. The tooltip uses `ChartTooltipContent` with `hideLabel` and `nameKey="device"`.

## Usage

```tsx
import { PieChartDonutText } from "@/components/charts/pie-chart-donut-text"
```

```tsx
<PieChartDonutText />
```

Replace `chartData` with your categories, giving each record a `fill` of `var(--color-<key>)`, and add a matching `chartConfig` entry with `label` and `color` for every key. Keep `dataKey` and `nameKey` on `Pie` pointed at your value and category fields, and update the tooltip's `nameKey` if you rename the category field. `total` is derived from `chartData`, so the center value updates automatically; change the `users` caption beneath it to describe your unit.

## Notes

The overlay uses `pointer-events-none` so hovering the center doesn't block the slices' tooltips. `accessibilityLayer` is set on `PieChart`, so the chart is focusable and arrow keys move between slices.
