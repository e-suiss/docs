```tsx title="components/charts/radial-chart-label.tsx"
"use client"

import { LabelList, RadialBar, RadialBarChart } from "recharts"

import {
  Card,
  CardContent,
  CardDescription,
  CardHeader,
  CardTitle,
} from "@/components/ui/card"
import {
  type ChartConfig,
  ChartContainer,
  ChartTooltip,
  ChartTooltipContent,
} from "@/components/ui/chart"

const chartData = [
  { device: "phone", users: 275, fill: "var(--color-phone)" },
  { device: "laptop", users: 200, fill: "var(--color-laptop)" },
  { device: "tablet", users: 187, fill: "var(--color-tablet)" },
  { device: "watch", users: 173, fill: "var(--color-watch)" },
]

const chartConfig = {
  users: { label: "Users" },
  phone: { label: "Phone", color: "var(--blue)" },
  laptop: { label: "Laptop", color: "var(--teal)" },
  tablet: { label: "Tablet", color: "var(--indigo)" },
  watch: { label: "Watch", color: "var(--orange)" },
} satisfies ChartConfig

export function RadialChartLabel() {
  return (
    <Card variant="filled" size="sm">
      <CardHeader>
        <CardTitle>Labeled</CardTitle>
        <CardDescription>
          Each name sits at the start of its bar
        </CardDescription>
      </CardHeader>
      <CardContent>
        <ChartContainer
          config={chartConfig}
          className="aspect-auto h-52 w-full"
        >
          <RadialBarChart
            accessibilityLayer
            data={chartData}
            innerRadius="28%"
            outerRadius="100%"
            barSize={16}
            startAngle={90}
            endAngle={-200}
          >
            <ChartTooltip
              cursor={false}
              content={<ChartTooltipContent hideLabel nameKey="device" />}
            />
            <RadialBar
              dataKey="users"
              cornerRadius={10}
              background={{ fill: "var(--control)" }}
            >
              <LabelList
                dataKey="device"
                position="insideStart"
                className="fill-on-accent text-2xs font-semibold"
                formatter={(value) =>
                  chartConfig[String(value) as keyof typeof chartConfig]?.label
                }
              />
            </RadialBar>
          </RadialBarChart>
        </ChartContainer>
      </CardContent>
    </Card>
  )
}
```

</Example>

Uses `RadialBarChart` with `innerRadius="28%"`, `barSize={16}`, `startAngle={90}` and `endAngle={-200}`, and one `RadialBar` (`cornerRadius={10}`, `var(--control)` background) containing a `LabelList` on `device` with `position="insideStart"`. The label `formatter` looks up each device's `label` in `chartConfig`. `chartData` holds four `{ device, users, fill }` records, where each `fill` is `var(--color-<device>)`, and `chartConfig` defines `users` (label only) plus `phone`, `laptop`, `tablet` and `watch` colored `var(--blue)`, `var(--teal)`, `var(--indigo)` and `var(--orange)`. The tooltip uses `hideLabel` and `nameKey="device"`.

## Usage

```tsx
import { RadialChartLabel } from "@/components/charts/radial-chart-label"
```

```tsx
<RadialChartLabel />
```

Replace `chartData` with your categories, giving each a `fill` of `var(--color-<key>)`, and add a matching `chartConfig` entry with `label` and `color`. Update the tooltip's `nameKey` if you rename the `device` field. Labels come from `chartConfig` labels through the `LabelList` `formatter`; keep them short so they fit inside the bar, or increase `barSize`.

## Notes

Labels use the `fill-on-accent` class so they stay legible on top of the colored bars. `accessibilityLayer` is set on `RadialBarChart`, so the chart is focusable from the keyboard.
