```tsx title="components/charts/tooltip-chart-icons.tsx"
"use client"

import { DesktopIcon, DeviceMobileIcon } from "@phosphor-icons/react"
import { Bar, BarChart, XAxis } from "recharts"

import {
  Card,
  CardContent,
  CardDescription,
  CardHeader,
  CardTitle,
} from "@/components/ui/card"
import {
  type ChartConfig,
  ChartContainer,
  ChartTooltip,
  ChartTooltipContent,
} from "@/components/ui/chart"

const chartData = [
  { month: "January", desktop: 186, mobile: 80 },
  { month: "February", desktop: 305, mobile: 200 },
  { month: "March", desktop: 237, mobile: 120 },
  { month: "April", desktop: 173, mobile: 190 },
  { month: "May", desktop: 209, mobile: 130 },
  { month: "June", desktop: 264, mobile: 140 },
]

const chartConfig = {
  desktop: { label: "Desktop", color: "var(--blue)", icon: DesktopIcon },
  mobile: { label: "Mobile", color: "var(--teal)", icon: DeviceMobileIcon },
} satisfies ChartConfig

export function TooltipChartIcons() {
  return (
    <Card variant="filled" size="sm">
      <CardHeader>
        <CardTitle>With icons</CardTitle>
        <CardDescription>An icon instead of a color</CardDescription>
      </CardHeader>
      <CardContent>
        <ChartContainer
          config={chartConfig}
          className="aspect-auto h-52 w-full"
        >
          <BarChart
            accessibilityLayer
            data={chartData}
            margin={{ top: 70, right: 4, left: 4, bottom: 0 }}
          >
            <XAxis
              dataKey="month"
              tickLine={false}
              axisLine={false}
              tickMargin={8}
              interval={0}
              tickFormatter={(value: string) => value.slice(0, 3)}
            />
            <ChartTooltip
              defaultIndex={3}
              cursor={false}
              content={<ChartTooltipContent />}
            />
            <Bar
              dataKey="desktop"
              stackId="a"
              fill="var(--color-desktop)"
              radius={[0, 0, 4, 4]}
              maxBarSize={30}
            />
            <Bar
              dataKey="mobile"
              stackId="a"
              fill="var(--color-mobile)"
              radius={[4, 4, 0, 0]}
              maxBarSize={30}
            />
          </BarChart>
        </ChartContainer>
      </CardContent>
    </Card>
  )
}
```

</Example>

Uses a `BarChart` with two `Bar` elements, `desktop` and `mobile`, stacked with `stackId="a"` (`maxBarSize={30}`, outer corners rounded), and an `XAxis` on `month` abbreviated to three letters. `chartData` holds six months of `{ month, desktop, mobile }` records, and `chartConfig` colors `desktop` with `var(--blue)` and `mobile` with `var(--teal)`. Each `chartConfig` entry also sets an `icon` (`DesktopIcon` and `DeviceMobileIcon` from `@phosphor-icons/react`). When an entry has an `icon`, `ChartTooltipContent` renders it instead of the indicator. `ChartTooltip` sets `defaultIndex={3}` so the tooltip is visible on load, and `cursor={false}`; the chart's top `margin` of 70 leaves room for it above the bars.

## Installation

### CLI

```bash title="Terminal"
npx @esuiss/ui@latest add charts tooltip-chart-icons
```

1. Install the dependencies:

```bash title="Terminal"

```

2. Add the pieces it builds on: [Card](/docs/ui/components/card), [Chart](/docs/ui/components/chart).

## Usage

```tsx
import { TooltipChartIcons } from "@/components/charts/tooltip-chart-icons"
```

```tsx
<TooltipChartIcons />
```

Add an `icon` component to any `chartConfig` entry to show it in the tooltip; entries without one fall back to the color indicator. Remove `defaultIndex` if the tooltip should appear only on hover or focus.

## Notes

`ChartLegendContent` also renders `chartConfig` icons unless `hideIcon` is set. `accessibilityLayer` is set on the chart, so the plot is focusable and arrow keys move the tooltip between categories.
