  ItemContent,
  ItemDescription,

  ItemSeparator,

  { value: "roles", label: "Roles", icon: <ShieldIcon /> },

    ],
  },
]

const roles = ["Editor", "Support", "Viewer"] as const

type Role = (typeof roles)[number]

const members: Record<Role, number> = { Editor: 2, Support: 2, Viewer: 1 }

const defaults: Record<Role, boolean[]> = {

  Support: [false, false, false, true, false],

const clone = (value: Record<Role, boolean[]>) => structuredClone(value)

export function RolesEditor() {

      items={nav}
      defaultValue="roles"
      header={
        <p className="flex items-center gap-2 px-2 pt-1 text-sm font-semibold">
          <span className="font-bold">suiss</span> Admin
        </p>
      }
    >
      <div className="flex items-center px-5 pt-4 md:hidden">
        <AppShellTrigger />
      </div>
      <SplitView
        defaultColumn="list"
        className="flex-1 rounded-none border-0 max-md:p-4 md:min-h-150"
      >
        <SplitViewList

        >
          {roles.map((name) => (
            <SplitViewItem
              key={name}
              isActive={name === role}
              onClick={() => setRole(name)}
              className="group/role py-1.5"
            >
              <span className="text-sm">{name}</span>
              <span className="text-xs text-label-secondary">
                {state[name].filter(Boolean).length} permissions
              </span>
            </SplitViewItem>
          ))}
          <Button
            size="sm"
            variant="secondary"
            className="mt-2.5 gap-1 self-start"
          >
            <PlusIcon weight="bold" className="size-3.25" />
            New Role
          </Button>
        </SplitViewList>
        <SplitViewDetail title={role} className="md:p-6 [&>h2]:sr-only">
          <div
            key={role}
            className="flex flex-col gap-4.5 transition-[opacity,translate] duration-800 ease-[cubic-bezier(0.32,0.72,0,1)] starting:translate-y-1.5 starting:opacity-0 motion-reduce:transition-none"
          >
            <div className="flex flex-wrap items-start gap-3">
              <div className="flex flex-1 flex-col">
                <p className="text-3xl font-semibold tracking-tight">{role}</p>
                <p className="text-sm text-label-secondary">
                  {members[role]}{" "}
                  {members[role] === 1 ? "user has" : "users have"} this role
                </p>
              </div>

                <div className="flex gap-2">
                  <Button
                    size="sm"
                    variant="secondary"
                    onClick={() => setState(clone(base))}
                  >
                    Revert
                  </Button>
                  <Button size="sm" onClick={() => setBase(clone(state))}>
                    Save
                  </Button>
                </div>
              )}
            </div>
            {groups.map((group) => (
              <section key={group.name} className="flex flex-col gap-2">
                <h2 className="px-1 text-sm font-semibold">{group.name}</h2>
                <ItemGroup variant="inset">
                  {group.permissions.map((permission, position) => {
                    const slot = index++
                    const id = `roles-editor-${slot}`
                    return (
                      <React.Fragment key={permission.name}>
                        {position > 0 && <ItemSeparator />}
                        <Item size="sm" role="listitem">
                          <ItemContent className="gap-0">

                            </ItemTitle>
                            <ItemDescription className="text-xs">

                                  [role]: current[role].map((item, i) =>
                                    i === slot ? value : item

                    )
                  })}
                </ItemGroup>
              </section>
            ))}
          </div>
        </SplitViewDetail>
      </SplitView>
    </AppShell>
  )
}

The block combines two patterns: `AppShell` provides the admin navigation, and inside it a `SplitView` places the roles in `SplitViewList` (as `SplitViewItem` rows with a permission count, plus a New Role `Button`) and the selected role in `SplitViewDetail`. The detail shows the role name, how many users have it, and permission groups built from `ItemGroup`, `Item`, `ItemContent`, `ItemTitle`, `ItemDescription`, `ItemActions`, and `ItemSeparator`, with a `Switch` for each permission. Revert and Save buttons appear only when there are unsaved changes. Below `md:` the split view shows one column at a time, starting with the list.

## Usage

```tsx
import { RolesEditor } from "@/components/blocks/roles-editor"
```

```tsx
<RolesEditor />
```

Edit `groups` for the permission groups (each permission has a `name` and `description`), `roles` for the role names, `members` for the user count per role, and `defaults` for each role's initial switches. The `defaults` arrays are positional: each boolean maps to a permission in the order it appears across all groups, so keep their length in step with `groups`. The selected role, the working `state`, and the saved `base` are held in `React.useState`; changes are detected by comparing the two. Replace the Save handler, which only copies `state` into `base`, with a call to your API, and give New Role a handler.

## Notes

Each permission name is a `label` tied to its `Switch` through `htmlFor` and a generated `id`. Permission rows use `role="listitem"` inside the list role of `ItemGroup`. The detail title is visually hidden with `sr-only`, while the list title is shown from `md:` up. The `AppShellTrigger` sits in a wrapper hidden at `md:`, and the trigger renders nothing on mobile-width viewports. The detail's entry animation is disabled with `motion-reduce:`.
