```tsx
import * as React from "react"
import { Pagination, PaginationContent, PaginationEllipsis, PaginationItem, PaginationLink, PaginationNext, PaginationPrevious, PaginationSummary } from "@/components/ui/pagination"
import { Table, TableBody, TableCell, TableHead, TableHeader, TableRow } from "@/components/ui/table"

const places = [
  "Table 7",
  "Table 4",
  "Takeaway",
  "Table 12",
  "Pickup",
  "Garden 3",
]

const currency = new Intl.NumberFormat("tr-TR", {
  style: "currency",
  currency: "TRY",
})

const orders = Array.from({ length: 62 }, (_, index) => {
  const id = 1100 - index
  return {
    id,
    place: places[(id * 7) % places.length],
    amount: ((id * 379) % 900) + 150,
  }
})

function pageRange(page: number, count: number) {
  if (count <= 7) return Array.from({ length: count }, (_, index) => index + 1)
  const start = Math.max(1, Math.min(page - 2, count - 6))
  const pages: (number | "ellipsis")[] = Array.from(
    { length: 5 },
    (_, index) => start + index
  )
  if (pages.at(-1) !== count) pages.push("ellipsis", count)
  return pages
}

function go(setPage: (page: number) => void, page: number, disabled: boolean) {
  return {
    href: "#",
    "aria-disabled": disabled || undefined,
    onClick: (event: React.MouseEvent) => {
      event.preventDefault()
      if (!disabled) setPage(page)
    },
  }
}

function Orders({
  page,
  perPage,
  children,
}: {
  page: number
  perPage: number
  children: React.ReactNode
}) {
  const rows = orders.slice((page - 1) * perPage, page * perPage)

  return (
    <div className="flex w-136 flex-col gap-4">
      <Table>
        <TableHeader>
          <TableRow>
            <TableHead>Order</TableHead>
            <TableHead>Table</TableHead>
            <TableHead className="text-end">Amount</TableHead>
          </TableRow>
        </TableHeader>
        <TableBody>
          {rows.map((order) => (
            <TableRow key={order.id}>
              <TableCell className="tabular-nums">#{order.id}</TableCell>
              <TableCell className="text-label-secondary">
                {order.place}
              </TableCell>
              <TableCell className="text-end tabular-nums">
                {currency.format(order.amount)}
              </TableCell>
            </TableRow>
          ))}
        </TableBody>
      </Table>
      {children}
    </div>
  )
}

function NumberedExample() {
  const [page, setPage] = React.useState(2)
  const perPage = 5
  const count = Math.ceil(orders.length / perPage)

  return (
    <Orders page={page} perPage={perPage}>
      <Pagination className="justify-between">
        <PaginationSummary>
          {(page - 1) * perPage + 1}–{Math.min(page * perPage, orders.length)} /{" "}
          {orders.length}
        </PaginationSummary>
        <PaginationContent>
          <PaginationItem>
            <PaginationPrevious
              text=""
              {...go(setPage, page - 1, page === 1)}
            />
          </PaginationItem>
          {pageRange(page, count).map((entry, index) =>
            entry === "ellipsis" ? (
              <PaginationItem key={`ellipsis-${index}`}>
                <PaginationEllipsis />
              </PaginationItem>
            ) : (
              <PaginationItem key={entry}>
                <PaginationLink
                  isActive={entry === page}
                  {...go(setPage, entry, false)}
                >
                  {entry}
                </PaginationLink>
              </PaginationItem>
            )
          )}
          <PaginationItem>
            <PaginationNext
              text=""
              {...go(setPage, page + 1, page === count)}
            />
          </PaginationItem>
        </PaginationContent>
      </Pagination>
    </Orders>
  )
}

export function PaginationExample() {
  return (
    <NumberedExample />
  )
}
```

Pagination is a set of link primitives you compose around your own page state. `Pagination` is the `nav` landmark. Inside it, `PaginationContent` is a list of `PaginationItem` entries, each holding a `PaginationLink`, `PaginationPrevious`, `PaginationNext`, `PaginationFirst`, `PaginationLast` or `PaginationEllipsis`.

For compact toolbars, `PaginationGroup` joins arrow links into one segmented control, `PaginationInput` lets the user type a page number, and `PaginationSummary` shows a range such as `6–10 / 62`. The component does not calculate page ranges; you decide which links to render.

```tsx title="components/ui/pagination.tsx"
  )

  text = "Previous",
```

## Usage


```tsx
<NumberedExample />
```

Links render as `a` elements, so pass `href` for real routes, or `href="#"` with an `onClick` that calls `preventDefault()` for client-side state. Mark the current page with `isActive` on `PaginationLink`. To disable an edge link, set `aria-disabled` on it; it becomes non-interactive and dimmed.

`PaginationLink` takes a `size` from the Button sizes and defaults to `icon-sm`. Set `text=""` on `PaginationPrevious` or `PaginationNext` for icon-only arrows.

```tsx
  const [page, setPage] = React.useState(3)
  const [draft, setDraft] = React.useState("3")
  const perPage = 5
  const count = Math.ceil(orders.length / perPage)

  const change = (next: number) => {
    setPage(next)
    setDraft(String(next))
  }

  const commit = () => {
    const next = Number.parseInt(draft, 10)
    if (Number.isNaN(next)) return setDraft(String(page))
    change(Math.min(Math.max(next, 1), count))
  }

  return (
    <Orders page={page} perPage={perPage}>
      <Pagination className="justify-between">
        <div className="flex items-center gap-2 text-sm text-label-secondary">
          <label htmlFor="orders-page">Page</label>
          <PaginationInput
            id="orders-page"
            value={draft}
            onChange={(event) => setDraft(event.target.value)}
            onBlur={commit}
            onKeyDown={(event) => event.key === "Enter" && commit()}
          />
          <span className="tabular-nums">/ {count}</span>
        </div>
        <PaginationGroup>
          <PaginationPrevious text="" {...go(change, page - 1, page === 1)} />
          <PaginationNext text="" {...go(change, page + 1, page === count)} />
        </PaginationGroup>
```

## API Reference

### Pagination

The `nav` landmark that contains the pagination controls.

Accepts every prop of `React.ComponentProps<"nav">`.

### PaginationContent

The list that holds the pagination items.

Accepts every prop of `React.ComponentProps<"ul">`.

### PaginationItem

A list item wrapper for one link or ellipsis.

Accepts every prop of `React.ComponentProps<"li">`.

### PaginationLink

A page link styled as a button, highlighted when active.

Accepts every prop of `PaginationLinkProps`.

### PaginationPrevious

A link to the previous page with a leading caret.

Accepts every prop of `React.ComponentProps<typeof PaginationLink>`.

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `text` | `string` | `"Previous"` | The visible label. A custom value also becomes the accessible name; an empty string renders an icon-only link. |

### PaginationNext

A link to the next page with a trailing caret.

Accepts every prop of `React.ComponentProps<typeof PaginationLink>`.

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `text` | `string` | `"Next"` | The visible label. A custom value also becomes the accessible name; an empty string renders an icon-only link. |

### PaginationFirst

An icon link to the first page.

Accepts every prop of `React.ComponentProps<typeof PaginationLink>`.

### PaginationLast

An icon link to the last page.

Accepts every prop of `React.ComponentProps<typeof PaginationLink>`.

### PaginationEllipsis

A placeholder that indicates skipped pages.

Accepts every prop of `React.ComponentProps<"span">`.

### PaginationSummary

Secondary text that summarizes the current range.

Accepts every prop of `React.ComponentProps<"p">`.

### PaginationGroup

Joins several links into one segmented control.

Accepts every prop of `React.ComponentProps<"div">`.

### PaginationInput

A numeric text field for jumping to a page.

Accepts every prop of `React.ComponentProps<"input">`.

## Accessibility

- `Pagination` renders a `nav` with `aria-label="pagination"`.
- The active `PaginationLink` gets `aria-current="page"`.
- Arrow links have built-in accessible names: `Go to previous page`, `Go to next page`, `Go to first page` and `Go to last page`. Custom `text` on previous or next replaces the name.
- `PaginationEllipsis` is hidden from assistive technology.
- `PaginationGroup` has `role="group"`.
- `PaginationInput` has no built-in label; associate one with `htmlFor` or `aria-label`.

## Notes

- Styling hooks: `data-slot` values `pagination`, `pagination-content`, `pagination-item`, `pagination-link`, `pagination-ellipsis`, `pagination-summary`, `pagination-group` and `pagination-input`. Active links expose `data-active`.
- Active links use the `secondary` button variant and others use `ghost`.
- Previous and next text is hidden below the `sm` breakpoint, and the carets flip in right-to-left layouts.
- `PaginationInput` uses `inputMode="numeric"` to show a number keypad on touch devices.
