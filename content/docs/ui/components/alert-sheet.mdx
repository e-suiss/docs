Alert Sheet is an iOS-style action sheet built on Base UI's dialog primitives. An `AlertSheet` root contains an `AlertSheetTrigger` and an `AlertSheetContent`, which renders its own portal and scrim overlay. Inside, an `AlertSheetGroup` holds an optional `AlertSheetHeader` (with `AlertSheetTitle` and `AlertSheetDescription`) and one or more `AlertSheetAction` rows. A separate `AlertSheetCancel` sits below the group.

Use it to offer a short list of actions tied to the current context, especially on touch devices. For a centered confirmation, use Alert Dialog.

"use client"

import { AlertDialog as AlertSheetPrimitive } from "@base-ui/react/alert-dialog"
import { Dialog as DialogPrimitive } from "@base-ui/react/dialog"
import { cva, type VariantProps } from "class-variance-authority"
import { cn } from "cn"
import * as React from "react"

function AlertSheet({
  dismissible = true,
  ...props
}: AlertSheetPrimitive.Root.Props & { dismissible?: boolean }) {
  if (dismissible) {
    return <DialogPrimitive.Root data-slot="alert-sheet" {...props} />
  }
  return <AlertSheetPrimitive.Root data-slot="alert-sheet" {...props} />
}

function AlertSheetTrigger({ ...props }: AlertSheetPrimitive.Trigger.Props) {
  return (
    <AlertSheetPrimitive.Trigger data-slot="alert-sheet-trigger" {...props} />
  )
}

function AlertSheetPortal({ ...props }: AlertSheetPrimitive.Portal.Props) {
  return (
    <AlertSheetPrimitive.Portal data-slot="alert-sheet-portal" {...props} />
  )
}

function AlertSheetOverlay({
  className,
  ...props
}: AlertSheetPrimitive.Backdrop.Props) {
  return (
    <AlertSheetPrimitive.Backdrop
      data-slot="alert-sheet-overlay"
      className={cn(
        "fixed inset-0 isolate z-50 bg-scrim transition-opacity duration-300 data-ending-style:opacity-0 data-starting-style:opacity-0",
        className
      )}
      {...props}
    />
  )
}

function AlertSheetContent({
  className,
  initialFocus,
  ref,
  ...props
}: AlertSheetPrimitive.Popup.Props) {
  const popupRef = React.useRef<HTMLDivElement | null>(null)

  const setRefs = (node: HTMLDivElement | null) => {
    popupRef.current = node
    if (typeof ref === "function") ref(node)
    else if (ref) ref.current = node
  }

  const focusCancel = () =>
    popupRef.current?.querySelector<HTMLElement>(
      '[data-slot="alert-sheet-cancel"]'
    ) ?? true

  return (
    <AlertSheetPortal>
      <AlertSheetOverlay />
      <AlertSheetPrimitive.Popup
        data-slot="alert-sheet-content"
        ref={setRefs}
        initialFocus={initialFocus ?? focusCancel}
        className={cn(
          "fixed inset-x-0 bottom-0 z-50 mx-auto flex w-full max-w-md flex-col gap-2 p-2 pb-[max(--spacing(2),env(safe-area-inset-bottom))] text-center outline-none transition-transform duration-300 ease-[cubic-bezier(0.32,0.72,0,1)] data-ending-style:translate-y-full data-starting-style:translate-y-full",
          className
        )}
        {...props}
      />
    </AlertSheetPortal>
  )
}

function AlertSheetGroup({ className, ...props }: React.ComponentProps<"div">) {
  return (
    <div
      data-slot="alert-sheet-group"
      className={cn(
        "flex flex-col overflow-hidden rounded-2xl bg-surface-raised text-label *:col-span-full *:not-first:border-t has-[>button>svg]:grid has-[>button>svg]:grid-cols-[1fr_auto_auto_1fr]",
        className
      )}
      {...props}
    />
  )
}

function AlertSheetHeader({
  className,
  ...props
}: React.ComponentProps<"div">) {
  return (
    <div
      data-slot="alert-sheet-header"
      className={cn("flex flex-col gap-1 p-4", className)}
      {...props}
    />
  )
}

function AlertSheetTitle({
  className,
  ...props
}: AlertSheetPrimitive.Title.Props) {
  return (
    <AlertSheetPrimitive.Title
      data-slot="alert-sheet-title"
      className={cn(
        "text-sm font-semibold text-label-secondary text-balance",
        className
      )}
      {...props}
    />
  )
}

function AlertSheetDescription({
  className,
  ...props
}: AlertSheetPrimitive.Description.Props) {
  return (
    <AlertSheetPrimitive.Description
      data-slot="alert-sheet-description"
      className={cn("text-sm text-label-secondary text-balance", className)}
      {...props}
    />
  )
}

const alertSheetActionVariants = cva(
  "flex min-h-14 w-full cursor-pointer items-center justify-center gap-2 px-4 text-lg has-[>svg]:grid has-[>svg]:grid-cols-subgrid has-[>svg]:gap-x-3 has-[>svg]:text-start [&>svg]:col-start-2 [&_svg:not([class*='size-'])]:size-5 outline-none hover:bg-[color-mix(in_oklab,var(--surface-raised),var(--label)_5%)] focus-visible:focus-ring active:bg-[color-mix(in_oklab,var(--surface-raised),var(--label)_10%)] disabled:pointer-events-none disabled:text-label-quaternary [--focus-ring-offset:-3px]",
  {
    variants: {
      variant: {
        default: "text-link",
        destructive: "text-danger",
      },
    },
    defaultVariants: {
      variant: "default",
    },
  }
)

function AlertSheetAction({
  className,
  variant,
  ...props
}: AlertSheetPrimitive.Close.Props &
  VariantProps<typeof alertSheetActionVariants>) {
  return (
    <AlertSheetPrimitive.Close
      data-slot="alert-sheet-action"
      className={cn(alertSheetActionVariants({ variant }), className)}
      {...props}
    />
  )
}

function AlertSheetCancel({
  className,
  ...props
}: AlertSheetPrimitive.Close.Props) {
  return (
    <AlertSheetPrimitive.Close
      data-slot="alert-sheet-cancel"
      className={cn(
        alertSheetActionVariants(),
        "rounded-2xl bg-surface-raised font-semibold",
        className
      )}
      {...props}
    />
  )
}

## Usage

```tsx
import { AlertSheet, AlertSheetTrigger, AlertSheetPortal, AlertSheetOverlay, AlertSheetContent, AlertSheetGroup, AlertSheetHeader, AlertSheetTitle, AlertSheetDescription, AlertSheetAction, AlertSheetCancel } from "@/components/ui/alert-sheet"
```

```tsx
<div className="flex min-h-svh items-center justify-center">
  <AlertSheet>
    <AlertSheetTrigger render={<Button variant="outline" />}>
      Delete order
    </AlertSheetTrigger>
    <AlertSheetContent>
      <AlertSheetGroup>
        <AlertSheetHeader>
          <AlertSheetTitle className="sr-only">Delete order</AlertSheetTitle>
          <AlertSheetDescription>
            Order #1042 will be permanently deleted. This action cannot be
            undone.
          </AlertSheetDescription>
        </AlertSheetHeader>
        <AlertSheetAction variant="destructive">
          Delete Order
        </AlertSheetAction>
      </AlertSheetGroup>
      <AlertSheetCancel>Cancel</AlertSheetCancel>
    </AlertSheetContent>
  </AlertSheet>
</div>
```

Every sheet needs an `AlertSheetTitle` for its accessible name. If the title should not be visible, hide it with `className="sr-only"`.

By default the sheet is `dismissible` and behaves like a regular dialog. Set `dismissible={false}` to render it as an alert dialog instead.

When the sheet opens, focus moves to `AlertSheetCancel` unless you pass your own `initialFocus` to `AlertSheetContent`. Every action and the cancel button close the sheet. Control the open state with `open` and `onOpenChange`, or use `defaultOpen`.

## API Reference

### AlertSheet

The root that manages the open state and dismissal behavior.

Accepts every prop of `AlertSheetPrimitive.Root.Props`.

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `dismissible` | `boolean` | `true` | When `true`, renders as a dialog; when `false`, renders as an alert dialog. |

### AlertSheetTrigger

The control that opens the sheet.

Accepts every prop of `AlertSheetPrimitive.Trigger.Props`.

### AlertSheetPortal

Renders the sheet into a portal outside the DOM hierarchy.

Accepts every prop of `AlertSheetPrimitive.Portal.Props`.

### AlertSheetOverlay

The full-screen scrim behind the sheet.

Accepts every prop of `AlertSheetPrimitive.Backdrop.Props`.

### AlertSheetContent

The bottom-anchored container, rendered with its own portal and overlay.

Accepts every prop of `AlertSheetPrimitive.Popup.Props`.

### AlertSheetGroup

A rounded surface that stacks the header and actions with dividers.

Accepts every prop of `React.ComponentProps<"div">`.

### AlertSheetHeader

A padded wrapper for the title and description.

Accepts every prop of `React.ComponentProps<"div">`.

### AlertSheetTitle

The heading that names the sheet.

Accepts every prop of `AlertSheetPrimitive.Title.Props`.

### AlertSheetDescription

Supporting text that describes the sheet.

Accepts every prop of `AlertSheetPrimitive.Description.Props`.

### AlertSheetAction

A full-width row button that performs an action and closes the sheet.

Accepts every prop of `AlertSheetPrimitive.Close.Props`.

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `variant` | `"default" \| "destructive"` | `"default"` | Text color of the action. `default` uses the link color; `destructive` uses the danger color. |

### AlertSheetCancel

A separate, bold button that dismisses the sheet.

Accepts every prop of `AlertSheetPrimitive.Close.Props`.

## Accessibility

- With `dismissible` (the default) the popup has the `dialog` role; with `dismissible={false}` it has the `alertdialog` role.
- The popup is named by `AlertSheetTitle` and described by `AlertSheetDescription`.
- Focus moves to the cancel button on open and returns to the trigger on close.
- `Escape` closes the sheet.

## Notes

- Styling hooks: `data-slot` values from `alert-sheet` through `alert-sheet-cancel`.
- The sheet slides with `translate-y-full` on `data-starting-style` and `data-ending-style`, and its bottom padding respects `env(safe-area-inset-bottom)`.
- When an action contains an icon, the group switches to a grid so icons and labels align across rows.
