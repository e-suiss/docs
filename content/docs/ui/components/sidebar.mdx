`SidebarProvider` holds the open state and wraps the whole layout. Inside it, `Sidebar` renders the panel and `SidebarInset` renders the main content beside it. The sidebar stacks `SidebarHeader`, a scrolling `SidebarContent` and `SidebarFooter`. Content is organized into `SidebarGroup` sections, each with a `SidebarGroupLabel`, an optional `SidebarGroupAction` and a `SidebarGroupContent`. Navigation lives in a `SidebarMenu` of `SidebarMenuItem` rows, each holding a `SidebarMenuButton` and optionally a `SidebarMenuAction`, `SidebarMenuBadge` or nested `SidebarMenuSub`.

`SidebarTrigger` and `SidebarRail` toggle the sidebar. On screens narrower than `48rem`, the sidebar renders inside a `Sheet` instead. Use `Sidebar` for the main navigation of an app; for bottom navigation on small screens, consider `TabBar`.

} from "@/components/ui/tooltip"
import { useIsMobile } from "@/hooks/use-mobile"
import { isMacPlatform } from "@/hooks/use-platform"

const SIDEBAR_COOKIE_NAME = "sidebar_state"
const SIDEBAR_COOKIE_MAX_AGE = 60 * 60 * 24 * 7
const SIDEBAR_WIDTH = "16rem"
const SIDEBAR_WIDTH_ICON = "3rem"
const SIDEBAR_KEYBOARD_SHORTCUT = "b"

type SidebarContextProps = {
  state: "expanded" | "collapsed"
  open: boolean
  setOpen: (open: boolean) => void
  openMobile: boolean
  setOpenMobile: (open: boolean) => void
  isMobile: boolean
  toggleSidebar: () => void
}

const SidebarContext = React.createContext<SidebarContextProps | null>(null)

function useSidebar() {
  const context = React.useContext(SidebarContext)
  if (!context) {
    throw new Error("useSidebar must be used within a SidebarProvider.")
  }

  return context
}

function SidebarProvider({
  defaultOpen = true,
  open: openProp,
  onOpenChange: setOpenProp,
  className,
  style,
  children,
  ...props
}: React.ComponentProps<"div"> & {
  defaultOpen?: boolean
  open?: boolean
  onOpenChange?: (open: boolean) => void
}) {
  const isMobile = useIsMobile()
  const [openMobile, setOpenMobile] = React.useState(false)

  const [_open, _setOpen] = React.useState(defaultOpen)
  const open = openProp ?? _open
  const setOpen = React.useCallback(
    (value: boolean | ((value: boolean) => boolean)) => {
      const openState = typeof value === "function" ? value(open) : value
      if (setOpenProp) {
        setOpenProp(openState)
      } else {
        _setOpen(openState)
      }

      document.cookie = `${SIDEBAR_COOKIE_NAME}=${openState}; path=/; max-age=${SIDEBAR_COOKIE_MAX_AGE}`
    },
    [setOpenProp, open]
  )

  const toggleSidebar = React.useCallback(() => {
    return isMobile ? setOpenMobile((open) => !open) : setOpen((open) => !open)
  }, [isMobile, setOpen])

  React.useEffect(() => {
    const handleKeyDown = (event: KeyboardEvent) => {
      if (
        event.key === SIDEBAR_KEYBOARD_SHORTCUT &&
        (isMacPlatform() ? event.metaKey : event.ctrlKey)
      ) {
        event.preventDefault()
        toggleSidebar()
      }
    }

    window.addEventListener("keydown", handleKeyDown)
    return () => window.removeEventListener("keydown", handleKeyDown)
  }, [toggleSidebar])

  const state = open ? "expanded" : "collapsed"

  const contextValue = React.useMemo<SidebarContextProps>(
    () => ({
      state,
      open,
      setOpen,
      isMobile,
      openMobile,
      setOpenMobile,
      toggleSidebar,
    }),
    [state, open, setOpen, isMobile, openMobile, toggleSidebar]
  )

  return (
    <SidebarContext.Provider value={contextValue}>
      <div
        data-slot="sidebar-wrapper"
        style={
          {
            "--sidebar-width": SIDEBAR_WIDTH,
            "--sidebar-width-icon": SIDEBAR_WIDTH_ICON,
            ...style,
          } as React.CSSProperties
        }
        className={cn(
          "flex min-h-svh w-full has-data-[variant=inset]:bg-surface-secondary",
          className
        )}
        {...props}
      >
        {children}
      </div>
    </SidebarContext.Provider>
  )
}

function Sidebar({

  collapsible = "offcanvas",
  className,
  children,
  dir,
  ...props

  collapsible?: "offcanvas" | "icon" | "none"
}) {
  const { isMobile, state, openMobile, setOpenMobile } = useSidebar()
  const direction = useDirection()

  if (collapsible === "none") {
    return (
      <div
        data-slot="sidebar"
        className={cn(
          "flex h-full w-(--sidebar-width) flex-col bg-surface-secondary text-label",

          </SheetHeader>
          <div className="flex h-full w-full flex-col">{children}</div>
        </SheetContent>
      </Sheet>
    )
  }

  return (
    <div
      className="group peer hidden text-label data-[side=right]:order-last md:block"
      data-state={state}
      data-collapsible={state === "collapsed" ? collapsible : ""}
      data-variant={variant}
      data-side={side}

            : "group-data-[collapsible=icon]:w-(--sidebar-width-icon) group-data-[side=left]:border-e group-data-[side=right]:border-s",

        {...props}

    </div>
  )
}

function SidebarMenuButton({
  render,
  isActive = false,
  variant,
  size = "default",
  tooltip,
  className,
  ...props
}: useRender.ComponentProps<"button"> &
  React.ComponentProps<"button"> & {
    isActive?: boolean
    tooltip?: string | React.ComponentProps<typeof TooltipContent>
  } & VariantProps<typeof sidebarMenuButtonVariants>) {
  const { isMobile, state } = useSidebar()
  const comp = useRender({
    defaultTagName: "button",
    props: mergeProps<"button">(
      {
        className: cn(sidebarMenuButtonVariants({ variant, size }), className),
      },
      props
    ),
    render: !tooltip ? (
      render
    ) : (
      <TooltipTrigger
        render={render ?? <button type="button" disabled={props.disabled} />}
      />
    ),
    state: {
      slot: "sidebar-menu-button",
      sidebar: "menu-button",
      size,
      active: isActive,
    },
  })

  if (!tooltip) {
    return comp
  }

  if (typeof tooltip === "string") {
    tooltip = {
      children: tooltip,
    }
  }

  return (
    <Tooltip>
      {comp}
      <TooltipContent
        side="right"
        align="center"
        hidden={state !== "collapsed" || isMobile}

## Usage

```tsx
import { SidebarProvider, Sidebar, SidebarTrigger, SidebarRail, SidebarInset, SidebarInput, SidebarHeader, SidebarFooter, SidebarSeparator, SidebarContent, SidebarGroup, SidebarGroupLabel, SidebarGroupAction, SidebarGroupContent, SidebarMenu, SidebarMenuItem, SidebarMenuButton, SidebarMenuAction, SidebarMenuBadge, SidebarMenuSkeleton, SidebarMenuSub, SidebarMenuSubItem, SidebarMenuSubButton } from "@/components/ui/sidebar"
```

```tsx
<AppShell side="left" variant="sidebar" collapsible="icon" />
```

Use `defaultOpen` on `SidebarProvider` for an uncontrolled sidebar, or `open` with `onOpenChange` to control it. Read and change state from any child with the `useSidebar` hook, which returns `state`, `open`, `setOpen`, `openMobile`, `setOpenMobile`, `isMobile` and `toggleSidebar`.

Set `collapsible="icon"` to shrink the sidebar to a rail of icons when collapsed, `offcanvas` to slide it out of view, or `none` to keep it fixed. Give each `SidebarMenuButton` a `tooltip` so its label stays available while collapsed to icons. Use the `render` prop on `SidebarMenuButton` to render it as a link.

## API Reference

### SidebarProvider

Holds the open state, provides it through context and wraps the layout.

Accepts every prop of `React.ComponentProps<"div">`.

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `defaultOpen` | `boolean` | `true` | Sets whether the sidebar starts expanded when uncontrolled. |
| `open` | `boolean` | — | Controls whether the sidebar is expanded. |
| `onOpenChange` | `(open: boolean) => void` | — | Called with the new state when the sidebar expands or collapses. |

### Sidebar

The sidebar panel; a sheet on mobile.

Accepts every prop of `React.ComponentProps<"div">`.

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `side` | `"left" \| "right"` | `"left"` | Places the sidebar on the left or right edge. |
| `variant` | `"sidebar" \| "floating" \| "inset"` | `"sidebar"` | Chooses a full-height `sidebar`, a rounded `floating` panel, or an `inset` layout. |
| `collapsible` | `"offcanvas" \| "icon" \| "none"` | `"offcanvas"` | Sets how the sidebar collapses: off canvas, to icons, or not at all. |

### SidebarTrigger

An icon button that toggles the sidebar.

Accepts every prop of `React.ComponentProps<typeof Button>`.

### SidebarRail

A thin hit area along the sidebar's edge that toggles it on click.

Accepts every prop of `React.ComponentProps<"button">`.

### SidebarInset

The `main` element for page content next to the sidebar.

Accepts every prop of `React.ComponentProps<"main">`.

### SidebarInput

A compact `Input` styled for use in the sidebar.

Accepts every prop of `React.ComponentProps<typeof Input>`.

### SidebarHeader

The top area, above the scrolling content.

Accepts every prop of `React.ComponentProps<"div">`.

### SidebarFooter

The bottom area, below the scrolling content.

Accepts every prop of `React.ComponentProps<"div">`.

### SidebarSeparator

A divider between sidebar sections.

Accepts every prop of `React.ComponentProps<typeof Separator>`.

### SidebarContent

The scrolling region that holds the groups.

Accepts every prop of `React.ComponentProps<"div">`.

### SidebarGroup

A section of related content.

Accepts every prop of `React.ComponentProps<"div">`.

### SidebarGroupLabel

The title of a group; it fades out when collapsed to icons.

Accepts every prop of `useRender.ComponentProps<"div">` and `React.ComponentProps<"div">`.

### SidebarGroupAction

A small button in the corner of a group, such as an add button.

Accepts every prop of `useRender.ComponentProps<"button">` and `React.ComponentProps<"button">`.

### SidebarGroupContent

The body of a group.

Accepts every prop of `React.ComponentProps<"div">`.

### SidebarMenu

A list of menu items.

Accepts every prop of `React.ComponentProps<"ul">`.

### SidebarMenuItem

A list item that holds a menu button and its extras.

Accepts every prop of `React.ComponentProps<"li">`.

### SidebarMenuButton

The clickable row for a destination or command.

Accepts every prop of `useRender.ComponentProps<"button">` and `React.ComponentProps<"button">`.

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `isActive` | `boolean` | `false` | Marks the button as the current item. |
| `tooltip` | `string \| React.ComponentProps<typeof TooltipContent>` | — | Text or `TooltipContent` props for a tooltip shown only while collapsed on desktop. |
| `variant` | `"default" \| "outline"` | `"default"` | Uses a plain style or an `outline` style with a background and border. |
| `size` | `"default" \| "sm" \| "lg"` | `"default"` | Sets the row height: `sm`, `default` or `lg`. |

### SidebarMenuAction

A small button at the end of a menu item, such as a more button.

Accepts every prop of `useRender.ComponentProps<"button">` and `React.ComponentProps<"button">`.

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `showOnHover` | `boolean` | `false` | On desktop, hides the action until its item is hovered or focused. |

### SidebarMenuBadge

A count or label shown at the end of a menu item.

Accepts every prop of `React.ComponentProps<"div">`.

### SidebarMenuSkeleton

A loading placeholder shaped like a menu item.

Accepts every prop of `React.ComponentProps<"div">`.

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `showIcon` | `boolean` | `false` | Adds a square placeholder for an icon. |

### SidebarMenuSub

A nested list of sub-items, indented with a leading line.

Accepts every prop of `React.ComponentProps<"ul">`.

### SidebarMenuSubItem

A list item inside a sub-menu.

Accepts every prop of `React.ComponentProps<"li">`.

### SidebarMenuSubButton

The clickable row of a sub-item, rendered as a link by default.

Accepts every prop of `useRender.ComponentProps<"a">` and `React.ComponentProps<"a">`.

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `size` | `"sm" \| "md"` | `"md"` | Sets the text size: `sm` or `md`. |
| `isActive` | `boolean` | `false` | Marks the sub-item as the current page. |

## Accessibility

- `SidebarTrigger` and `SidebarRail` are labelled `Toggle Sidebar`. The rail is removed from the tab order, so keep a `SidebarTrigger` in reach of keyboard users.
- Press `Command-B` on Apple platforms, or `Ctrl+B` elsewhere, to toggle the sidebar.
- On mobile, the sidebar is a dialog with a visually hidden title, `Sidebar`; `Escape` closes it.
- Menu-button tooltips appear only while the sidebar is collapsed, so labels stay available when only icons show.

## Notes

- Every toggle writes the state to a `sidebar_state` cookie for 7 days. Read it on the server to set `defaultOpen` and avoid a flash on load.
- Widths come from `--sidebar-width` (`16rem`) and `--sidebar-width-icon` (`3rem`), set on `SidebarProvider`; override them through its `style`.
- The sidebar element exposes `data-state` (`expanded` or `collapsed`), `data-collapsible`, `data-variant` and `data-side` for styling. Menu buttons set `data-active` and `data-size`.
- `useSidebar` throws if it's used outside a `SidebarProvider`.
- The panel uses `bg-surface-secondary`; active items use `bg-item-selected` with a `text-link` icon.
